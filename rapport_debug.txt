1. STATUT DES APIS & PROXIES CORS (BENCHMARK HTTP EN DIRECT)
--------------------------------------------------------------------------------
Les tests de requêtes HTTP en direct pour la recherche ('rocky fut') et l'extraction
ont donné les métriques suivantes depuis une connexion réelle :

[FONCTIONNELS - RAPIDES]
- Invidious API (ChocolateMoo53) :
  URL Streams : https://yt.chocolatemoo53.com/api/v1/videos/jB5puKULM9w
  URL Search  : https://yt.chocolatemoo53.com/api/v1/search?q=rocky%20fut
  Statut      : HTTP 200 OK | Temps de réponse flux : ~501 ms
  Données     : formatStreams (MP4 720p/360p), adaptiveFormats (MP4), hlsUrl direct
  Avatars     : authorThumbnails (résolu à 100%)
  Miniatures  : videoThumbnails[0].url avec fallback i.ytimg.com

- Piped API (Ducks Party) :
  URL    : https://pipedapi.ducks.party/search?q=rocky%20fut&filter=all
  Statut : HTTP 200 OK | Temps de réponse : ~224 ms
  Données: 19 vidéos fraîches de Rocky FUT (FC 25, FUT Champions, Drafts...)
  Avatars: uploaderAvatar mappé

- Piped API (Private Coffee) :
  URL    : https://api.piped.private.coffee/search?q=rocky%20fut&filter=all
  Statut : HTTP 200 OK | Temps de réponse : ~276 ms
  Données: 19 vidéos fraîches de Rocky FUT
  Avatars: uploaderAvatar mappé

- Embed YouTube NoCookie :
  URL    : https://www.youtube-nocookie.com/embed/jB5puKULM9w
  Statut : HTTP 200 OK | Temps de réponse : ~267 ms
  Usage  : Moteur principal sans blocage pare-feu en entreprise

- Embed Piped (Cloudflare) :
  URL    : https://cf.piped.video/embed/jB5puKULM9w
  Statut : HTTP 200 OK | Temps de réponse : ~152 ms
  Usage  : Fallback propre sans iframe YouTube

[INSTABLES / BLOQUÉS / TIMEOUT]
- Proxy CORS AllOrigins :
  URL    : https://api.allorigins.win/raw?url=...
  Statut : HTTP 403 Forbidden (Détection Cloudflare Turnstile / anti-bot actif)

- Proxy CORS CodeTabs :
  URL    : https://api.codetabs.com/v1/proxy/?quest=...
  Statut : Timeout (> 4000 ms) ou Erreur 504 sur les requêtes YouTube volumineuses

STRATÉGIE RETENUE :
Utilisation de Promise.any() en parallèle sur les 3 instances API Piped/Invidious
avec un timeout strict de 3500-4000 ms (AbortController). Le premier endpoint qui
répond fournit les résultats en < 300 ms. Zéro attente séquentielle bloquante.


2. GÉNÉRATION DU DOM DES CARTES VIDÉOS (renderGrid)
--------------------------------------------------------------------------------
Correction définitive : Remplacement intégral des template strings HTML inline
(qui cassaient les balises avec ', true)"> et créaient des erreurs de guillemets)
par une création 100% programmatique via document.createElement() et addEventListener().

function renderGrid(videos) {
      const grid = document.getElementById('video-grid');
      grid.innerHTML = '';
      window._videoRegistry = {};

      if (!videos || !videos.length) {
        grid.innerHTML = `
          <div class="empty-state">
            <p>Aucune vidéo disponible pour le moment.</p>
          </div>`;
        return;
      }

      videos.forEach(v => {
        if (!v || !v.id) return;
        const viewsText = formatViews(v.views);
        const durationText = formatDuration(v.duration);
        const videoObj = { ...v, viewsText, durationText };
        window._videoRegistry[v.id] = videoObj;

        const card = document.createElement('div');
        card.className = 'video-card';
        card.setAttribute('data-video-id', v.id);

        card.addEventListener('click', (e) => {
          if (e.target.closest('.channel-click-target')) return;
          playVideo(videoObj, true);
        });

        // 1. Thumbnail Wrapper
        const thumbWrapper = document.createElement('div');
        thumbWrapper.className = 'thumb-wrapper';

        const thumbImg = document.createElement('img');
        thumbImg.className = 'thumb-img';
        thumbImg.alt = v.title || '';
        thumbImg.loading = 'lazy';
        thumbImg.src = v.thumbnail || `https://i.ytimg.com/vi/${v.id}/hqdefault.jpg`;
        thumbImg.addEventListener('error', () => {
          thumbImg.src = `https://i.ytimg.com/vi/${v.id}/hqdefault.jpg`;
        });
        thumbWrapper.appendChild(thumbImg);

        if (durationText) {
          const durEl = document.createElement('span');
          durEl.className = 'video-duration';
          durEl.textContent = durationText;
          thumbWrapper.appendChild(durEl);
        }
        card.appendChild(thumbWrapper);

        // 2. Video Meta
        const metaDiv = document.createElement('div');
        metaDiv.className = 'video-meta';

        // Avatar
        const avatarBtn = document.createElement('div');
        avatarBtn.className = 'avatar-clickable channel-click-target';
        avatarBtn.title = `Accéder à la chaîne ${v.uploader || ''}`;
        avatarBtn.addEventListener('click', (e) => {
          e.stopPropagation();
          openChannel(v.channelId || '', v.uploader || '', v.avatar || '', true);
        });

        const avatarImg = document.createElement('img');
        avatarImg.className = 'channel-avatar';
        avatarImg.alt = '';
        avatarImg.loading = 'lazy';
        avatarImg.src = v.avatar || "data:image/svg+xml,<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 24 24' fill='%23333'><circle cx='12' cy='12' r='10'/></svg>";
        avatarBtn.appendChild(avatarImg);
        metaDiv.appendChild(avatarBtn);

        // Info
        const infoDiv = document.createElement('div');
        infoDiv.className = 'video-info';

        const titleEl = document.createElement('div');
        titleEl.className = 'video-title';
        titleEl.title = v.title || '';
        titleEl.textContent = v.title || '';
        infoDiv.appendChild(titleEl);

        const channelEl = document.createElement('div');
        channelEl.className = 'channel-title channel-click-target';
        channelEl.title = `Accéder à la chaîne ${v.uploader || ''}`;
        channelEl.textContent = v.uploader || 'YouTube';
        channelEl.addEventListener('click', (e) => {
          e.stopPropagation();
          openChannel(v.channelId || '', v.uploader || '', v.avatar || '', true);
        });
        infoDiv.appendChild(channelEl);

        const substatsEl = document.createElement('div');
        substatsEl.className = 'video-substats';
        if (viewsText) {
          const viewsSpan = document.createElement('span');
          viewsSpan.textContent = viewsText;
          substatsEl.appendChild(viewsSpan);
        }
        if (viewsText && v.uploadedText) {
          const dotSpan = document.createElement('span');
          dotSpan.textContent = ' • ';
          substatsEl.appendChild(dotSpan);
        }
        if (v.uploadedText) {
          const uploadSpan = document.createElement('span');
          uploadSpan.textContent = v.uploadedText;
          substatsEl.appendChild(uploadSpan);
        }
        infoDiv.appendChild(substatsEl);

        metaDiv.appendChild(infoDiv);
        card.appendChild(metaDiv);
        grid.appendChild(card);
      });
    }


3. FONCTION DE RECHERCHE EN DIRECT & TENDANCES (fetchLiveSearch)
--------------------------------------------------------------------------------
Recherche multi-sources ultra-rapide avec Promise.any(), mapping complet des avatars
(uploaderAvatar / authorThumbnails) et des miniatures (videoThumbnails / i.ytimg.com) :

async function fetchLiveSearch(query, signal) {
      if (!query || !query.trim()) return null;

      const endpoints = [
        { url: `https://pipedapi.ducks.party/search?q=${encodeURIComponent(query)}&filter=all`, type: 'piped' },
        { url: `https://api.piped.private.coffee/search?q=${encodeURIComponent(query)}&filter=all`, type: 'piped' },
        { url: `https://yt.chocolatemoo53.com/api/v1/search?q=${encodeURIComponent(query)}`, type: 'invidious' }
      ];

      const promises = endpoints.map(api => {
        return new Promise(async (resolve, reject) => {
          try {
            const res = await fetch(api.url, { signal });
            if (!res.ok) return reject(new Error(`HTTP ${res.status}`));
            const data = await res.json();
            const isInv = api.type === 'invidious';
            const rawItems = isInv ? data : (data.items || data);
            if (!Array.isArray(rawItems) || rawItems.length === 0) return reject(new Error("Empty"));

            const validVideos = rawItems.map(it => {
              const videoId = isInv ? (it.videoId || extractVideoId(it.url)) : (extractVideoId(it.url) || it.id);
              if (!videoId || videoId.length !== 11) return null;

              // Avatar
              const rawAvatar = isInv
                ? (Array.isArray(it.authorThumbnails) ? (it.authorThumbnails[it.authorThumbnails.length - 1]?.url || it.authorThumbnails[0]?.url || it.authorThumbnails[0]) : '')
                : (it.uploaderAvatar || '');
              const avatar = typeof rawAvatar === 'string' ? rawAvatar : (rawAvatar?.url || '');

              // Thumbnail avec fallback i.ytimg.com
              const rawThumb = isInv
                ? (it.videoThumbnails?.[0]?.url || it.thumbnail)
                : (it.thumbnail);
              const thumbnail = rawThumb || `https://i.ytimg.com/vi/${videoId}/hqdefault.jpg`;

              return {
                id: videoId,
                title: it.title || 'Vidéo YouTube',
                uploader: isInv ? it.author : it.uploaderName,
                channelId: isInv ? it.authorId : (it.uploaderUrl || '').replace('/channel/', ''),
                avatar: avatar,
                thumbnail: thumbnail,
                views: it.viewCount || it.views || '',
                uploadedText: it.publishedText || it.uploadedDate || '',
                duration: it.lengthSeconds || it.duration || ''
              };
            }).filter(Boolean);

            if (validVideos.length === 0) return reject(new Error("No video items"));
            resolve(validVideos);
          } catch (err) {
            reject(err);
          }
        });
      });

      try {
        return await Promise.any(promises);
      } catch (e) {
        // En secours : scraping YouTube via proxy CORS
        try {
          const targetUrl = `https://www.youtube.com/results?search_query=${encodeURIComponent(query)}`;
          const proxyUrl = `https://api.codetabs.com/v1/proxy/?quest=${encodeURIComponent(targetUrl)}`;
          const res = await fetch(proxyUrl, { signal });
          if (res.ok) {
            const html = await res.text();
            const parsed = parseYtInitialData(html);
            if (parsed && parsed.length > 0) return parsed;
          }
        } catch (e2) {}
      }
      return null;
    }


4. LECTEUR DIRECT MP4 CORRIGÉ (playDirectStream - PROMISE.ANY PARALLÈLE)
--------------------------------------------------------------------------------
Correction de la boucle for...of séquentielle : extraction en parallèle avec Promise.any()
sur 3 endpoints (Invidious Choco, Piped Ducks, Piped Coffee) avec timeout de 3500 ms.
Support complet de data.formatStreams (720p/360p combinés), data.adaptiveFormats et hlsUrl :

async function playDirectStream(videoId) {
      hideAllPlayerContainers();
      const directCont = document.getElementById('direct-player-container');
      const loading = document.getElementById('direct-loading-overlay');
      const loadingText = document.getElementById('direct-loading-text');

      directCont.style.display = 'block';
      if (loading) loading.style.display = 'flex';
      if (loadingText) loadingText.textContent = "Extraction du flux vidéo direct sans pub (Full HD / 720p)...";

      initPlyrPlayer();
      const videoEl = document.getElementById('plyr-video');

      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), 3500); // Timeout 3500ms

      const streamEndpoints = [
        { name: 'Invidious Choco API', url: `https://yt.chocolatemoo53.com/api/v1/videos/${videoId}`, type: 'invidious' },
        { name: 'Piped Ducks API', url: `https://pipedapi.ducks.party/streams/${videoId}`, type: 'piped' },
        { name: 'Piped Coffee API', url: `https://api.piped.private.coffee/streams/${videoId}`, type: 'piped' }
      ];

      const streamPromises = streamEndpoints.map(api => {
        return new Promise(async (resolve, reject) => {
          try {
            const res = await fetch(api.url, { signal: controller.signal });
            if (!res.ok) return reject(new Error(`${api.name}: HTTP ${res.status}`));
            const data = await res.json();

            let mp4 = null;
            let hls = data.hlsUrl || data.hls || null;

            if (api.type === 'invidious') {
              if (data.formatStreams && data.formatStreams.length > 0) {
                const sorted = [...data.formatStreams].sort((a, b) => (parseInt(b.resolution) || 0) - (parseInt(a.resolution) || 0));
                mp4 = sorted[0].url;
              } else if (data.adaptiveFormats && data.adaptiveFormats.length > 0) {
                const mp4Adaptive = data.adaptiveFormats.find(f => f.type && f.type.startsWith('video/mp4') && f.url);
                if (mp4Adaptive) mp4 = mp4Adaptive.url;
              }
            } else {
              // Piped
              const muxed = (data.videoStreams || []).filter(s => !s.videoOnly && s.url);
              if (muxed.length > 0) {
                muxed.sort((a, b) => (parseInt(b.quality) || 0) - (parseInt(a.quality) || 0));
                mp4 = muxed[0].url;
              } else if (data.videoStreams && data.videoStreams.length > 0) {
                const v = data.videoStreams.find(s => s.url);
                if (v) mp4 = v.url;
              }
            }

            if (mp4 || hls) {
              resolve({ mp4, hls, apiName: api.name });
            } else {
              reject(new Error(`${api.name}: aucun flux exploitable`));
            }
          } catch (err) {
            reject(err);
          }
        });
      });

      let streamResult = null;
      try {
        streamResult = await Promise.any(streamPromises);
      } catch (err) {
        console.warn("Échec extraction flux direct:", err);
      } finally {
        clearTimeout(timeoutId);
      }

      if (loading) loading.style.display = 'none';

      if (streamResult && (streamResult.mp4 || streamResult.hls)) {
        if (streamResult.hls && window.Hls && Hls.isSupported()) {
          if (hlsInstance) hlsInstance.destroy();
          hlsInstance = new Hls();
          hlsInstance.loadSource(streamResult.hls);
          hlsInstance.attachMedia(videoEl);
          hlsInstance.on(Hls.Events.MANIFEST_PARSED, () => {
            videoEl.play().catch(e => console.log("Autoplay:", e));
          });
        } else if (streamResult.mp4) {
          videoEl.src = streamResult.mp4;
          videoEl.load();
          videoEl.play().catch(e => console.log("Autoplay:", e));
        }
        showToast(`🎬 Flux Direct HTML5 actif (${streamResult.apiName})`);
      } else {
        showToast("⚡ Flux direct indisponible : bascule automatique sur NoCookie Anti-Pub");
        switchEngine('nocookie');
      }
    }

async function loadRelatedVideos(videoId, videoTitle) {
      const relGrid = document.getElementById('related-videos-grid');
      if (!relGrid) return;
      relGrid.innerHTML = `
        <div class="loading-container" style="grid-column: 1 / -1; padding: 20px 0;">
          <div class="spinner"></div>
          <p style="font-size: 0.85rem; color: var(--text-muted);">Chargement des vidéos associées...</p>
        </div>`;

      const controller = new AbortController();
      const tid = setTimeout(() => controller.abort(), 3500);

      const endpoints = [
        { name: 'Invidious Choco', url: `https://yt.chocolatemoo53.com/api/v1/videos/${videoId}`, type: 'invidious' },
        { name: 'Piped Ducks', url: `https://pipedapi.ducks.party/streams/${videoId}`, type: 'piped' },
        { name: 'Piped Coffee', url: `https://api.piped.private.coffee/streams/${videoId}`, type: 'piped' }
      ];

      const promises = endpoints.map(api => {
        return new Promise(async (resolve, reject) => {
          try {
            const res = await fetch(api.url, { signal: controller.signal });
            if (!res.ok) return reject(new Error(`HTTP ${res.status}`));
            const data = await res.json();
            const isInv = api.type === 'invidious';
            const list = isInv ? data.recommendedVideos : data.relatedStreams;
            if (!Array.isArray(list) || list.length === 0) return reject(new Error("Empty"));

            const mapped = list.map(item => {
              const relId = isInv ? item.videoId : extractVideoId(item.url || '');
              if (!relId || relId === videoId || relId.length !== 11) return null;
              const rawThumb = isInv ? (item.videoThumbnails?.[0]?.url || item.thumbnail) : item.thumbnail;
              return {
                id: relId,
                title: item.title || 'Vidéo associée',
                thumbnail: rawThumb || `https://i.ytimg.com/vi/${relId}/hqdefault.jpg`,
                uploader: isInv ? item.author : (item.uploaderName || 'YouTube'),
                channelId: isInv ? item.authorId : (item.uploaderUrl || '').replace('/channel/', ''),
                duration: item.lengthSeconds || item.duration || '',
                views: item.viewCount || item.views || ''
              };
            }).filter(Boolean);

            if (mapped.length === 0) return reject(new Error("No valid related"));
            resolve(mapped);
          } catch (e) {
            reject(e);
          }
        });
      });

      try {
        const liveRelated = await Promise.any(promises);
        clearTimeout(tid);
        if (liveRelated && liveRelated.length > 0) {
          currentRelatedVideos = liveRelated.slice(0, 12);
          renderRelatedGrid(currentRelatedVideos);
          return;
        }
      } catch (e) {
        clearTimeout(tid);
      }

      // Fallback local uniquement si échec réseau complet
      currentRelatedVideos = (FALLBACK_CATALOG.trending || []).filter(x => x.id !== videoId).slice(0, 8);
      renderRelatedGrid(currentRelatedVideos);
    }

function executePlayback(engine, videoId) {
      currentEngine = engine;
      renderEnginePills();

      const adTools = document.getElementById('nocookie-ad-tools');
      const workTip = document.getElementById('nocookie-work-tip');

      if (adTools) adTools.style.display = (engine === 'nocookie') ? 'flex' : 'none';
      if (workTip) workTip.style.display = (engine === 'nocookie') ? 'flex' : 'none';

      if (engine === 'direct') {
        playDirectStream(videoId);
      } else if (engine === 'nocookie') {
        loadYouTubeIframePlayer(videoId, 1);
      } else {
        hideAllPlayerContainers();
        const frame = document.getElementById('video-frame');
        frame.style.display = 'block';
        frame.src = ENGINES[engine] ? ENGINES[engine](videoId) : `https://cf.piped.video/embed/${videoId}?autoplay=1`;
      }
    }
