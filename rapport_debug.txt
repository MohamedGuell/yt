2. CODE CORRIGÉ : LECTEUR DIRECT MP4 AVEC PROXIFICATION &local=true (playDirectStream)

Fonction de réécriture d'URL avec proxy local (contourne le 403 Forbidden IP mismatch) :
function toProxiedUrl(rawUrl, instanceUrl) {
      if (!rawUrl) return null;
      try {
        const u = new URL(rawUrl);
        if (u.hostname.includes('googlevideo.com')) {
          const inst = new URL(instanceUrl);
          u.protocol = inst.protocol;
          u.host = inst.host;
        }
        if (!u.searchParams.has('local')) {
          u.searchParams.set('local', 'true');
        }
        return u.toString();
      } catch (e) {
        if (rawUrl.includes('?')) return `${rawUrl}&local=true`;
        return rawUrl;
      }
    }

Extraction parallèle Promise.any() avec timeout de 5000 ms sur 5 endpoints Invidious & Piped :
async function playDirectStream(videoId) {
      hideAllPlayerContainers();
      const directCont = document.getElementById('direct-player-container');
      const loading = document.getElementById('direct-loading-overlay');
      const loadingText = document.getElementById('direct-loading-text');

      directCont.style.display = 'block';
      if (loading) loading.style.display = 'flex';
      if (loadingText) loadingText.textContent = "Extraction du flux vidéo direct sans pub (Full HD / 720p)...";

      initPlyrPlayer();
      const videoEl = document.getElementById('plyr-video');

      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), 5000); // 5 secondes en parallèle

      const streamEndpoints = [
        { name: 'Invidious Choco', url: `https://yt.chocolatemoo53.com/api/v1/videos/${videoId}`, type: 'invidious', base: 'https://yt.chocolatemoo53.com' },
        { name: 'Invidious Nadeko', url: `https://inv.nadeko.net/api/v1/videos/${videoId}`, type: 'invidious', base: 'https://inv.nadeko.net' },
        { name: 'Invidious NerdVPN', url: `https://invidious.nerdvpn.de/api/v1/videos/${videoId}`, type: 'invidious', base: 'https://invidious.nerdvpn.de' },
        { name: 'Piped Ducks', url: `https://pipedapi.ducks.party/streams/${videoId}`, type: 'piped', base: 'https://pipedapi.ducks.party' },
        { name: 'Piped Coffee', url: `https://api.piped.private.coffee/streams/${videoId}`, type: 'piped', base: 'https://api.piped.private.coffee' }
      ];

      const streamPromises = streamEndpoints.map(api => {
        return new Promise(async (resolve, reject) => {
          try {
            const res = await fetch(api.url, { signal: controller.signal });
            if (!res.ok) return reject(new Error(`${api.name}: HTTP ${res.status}`));
            const data = await res.json();

            let mp4 = null;
            let hls = data.hlsUrl || data.hls || null;

            if (api.type === 'invidious') {
              if (data.formatStreams && data.formatStreams.length > 0) {
                const sorted = [...data.formatStreams].sort((a, b) => (parseInt(b.resolution) || 0) - (parseInt(a.resolution) || 0));
                mp4 = toProxiedUrl(sorted[0].url, api.base);
              } else if (data.adaptiveFormats && data.adaptiveFormats.length > 0) {
                const mp4Adaptive = data.adaptiveFormats.find(f => f.type && f.type.startsWith('video/mp4') && f.url);
                if (mp4Adaptive) mp4 = toProxiedUrl(mp4Adaptive.url, api.base);
              }
            } else {
              // Piped
              const muxed = (data.videoStreams || []).filter(s => !s.videoOnly && s.url);
              if (muxed.length > 0) {
                muxed.sort((a, b) => (parseInt(b.quality) || 0) - (parseInt(a.quality) || 0));
                mp4 = muxed[0].url;
              } else if (data.videoStreams && data.videoStreams.length > 0) {
                const v = data.videoStreams.find(s => s.url);
                if (v) mp4 = v.url;
              }
            }

            if (mp4 || hls) {
              resolve({ mp4, hls, apiName: api.name });
            } else {
              reject(new Error(`${api.name}: aucun flux exploitable`));
            }
          } catch (err) {
            reject(err);
          }
        });
      });

      let streamResult = null;
      try {
        streamResult = await Promise.any(streamPromises);
      } catch (err) {
        console.warn("Échec extraction flux direct:", err);
      } finally {
        clearTimeout(timeoutId);
      }

      if (loading) loading.style.display = 'none';

      if (streamResult && (streamResult.mp4 || streamResult.hls)) {
        let playStarted = false;
        if (streamResult.hls && window.Hls && Hls.isSupported()) {
          try {
            if (hlsInstance) hlsInstance.destroy();
            hlsInstance = new Hls();
            hlsInstance.loadSource(streamResult.hls);
            hlsInstance.attachMedia(videoEl);
            hlsInstance.on(Hls.Events.MANIFEST_PARSED, () => {
              videoEl.play().catch(e => console.log("Autoplay:", e));
            });
            playStarted = true;
          } catch (e) {
            console.warn("Erreur HLS:", e);
          }
        }
        
        if (!playStarted && streamResult.mp4) {
          videoEl.src = streamResult.mp4;
          videoEl.load();
          videoEl.play().catch(e => console.log("Autoplay:", e));
          playStarted = true;
        }

        if (playStarted) {
          showToast(`🎬 Flux Direct HTML5 actif (${streamResult.apiName})`);
          return;
        }
      }

      showToast("⚡ Flux direct indisponible : bascule automatique sur NoCookie Anti-Pub");
      switchEngine('nocookie');
    }


3. CODE AMÉLIORÉ : CONTOURNEMENT ANTI-PUB NOCOOKIE DYNAMIQUE (skipNoCookieAdByTimestamp)
--------------------------------------------------------------------------------
Lecture dynamique de ytPlayer.getCurrentTime() et saut instantané à start=Math.floor(time)+1 :
function skipNoCookieAdByTimestamp() {
      if (isAdReloading) return;
      isAdReloading = true;

      let resumeAt = Math.max(1, lastKnownTime || 1);
      try {
        if (ytPlayer && typeof ytPlayer.getCurrentTime === 'function') {
          const t = ytPlayer.getCurrentTime();
          if (t && t > 0) resumeAt = Math.floor(t);
        }
      } catch (e) {}

      const nextStart = resumeAt + 1;
      showToast(`⚡ Saut de pub : rechargement à ${formatTime(nextStart)}...`);

      try {
        if (ytPlayer && typeof ytPlayer.loadVideoById === 'function') {
          ytPlayer.loadVideoById({
            videoId: currentVideoId,
            startSeconds: nextStart
          });
          setTimeout(() => { isAdReloading = false; }, 1500);
          return;
        }
      } catch (e) {}

      loadYouTubeIframePlayer(currentVideoId, nextStart);
      setTimeout(() => { isAdReloading = false; }, 1500);
    }

function executePlayback(engine, videoId) {
      currentEngine = engine;
      renderEnginePills();

      const selectGlobal = document.getElementById('global-engine');
      if (selectGlobal) selectGlobal.value = engine;
      const selectPlayer = document.getElementById('player-engine-select');
      if (selectPlayer) selectPlayer.value = engine;

      const adTools = document.getElementById('nocookie-ad-tools');
      const workTip = document.getElementById('nocookie-work-tip');

      if (adTools) adTools.style.display = (engine === 'nocookie') ? 'flex' : 'none';
      if (workTip) workTip.style.display = (engine === 'nocookie') ? 'flex' : 'none';

      if (engine === 'direct') {
        playDirectStream(videoId);
      } else if (engine === 'nocookie') {
        loadYouTubeIframePlayer(videoId, 1);
      } else {
        hideAllPlayerContainers();
        const frame = document.getElementById('video-frame');
        frame.style.display = 'block';
        frame.src = ENGINES[engine] ? ENGINES[engine](videoId) : `https://www.youtube-nocookie.com/embed/${videoId}?autoplay=1&start=1&rel=0&iv_load_policy=3&playsinline=1&enablejsapi=1`;
      }
    }
